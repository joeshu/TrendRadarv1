## 三、网络层：URLSession + Combine  
  
### 3.1 NetworkManager  
  
```swift  
import Foundation  
import Combine  
  
enum NetworkError: Error {  
    case invalidURL  
    case invalidResponse  
    case decodingFailed(Error)  
    case serverError(Int)  
    case noData  
}  
  
final class NetworkManager {  
    static let shared = NetworkManager()  
      
    private let session: URLSession  
    private let decoder: JSONDecoder  
      
    private init() {  
        let config = URLSessionConfiguration.default  
        config.timeoutIntervalForRequest = 15  
        config.timeoutIntervalForResource = 30  
        config.requestCachePolicy = .reloadRevalidatingCacheData  
        self.session = URLSession(configuration: config)  
          
        self.decoder = JSONDecoder()  
        self.decoder.keyDecodingStrategy = .convertFromSnakeCase  
        self.decoder.dateDecodingStrategy = .iso8601  
    }  
      
    func request<T: Decodable>(_ endpoint: APIEndpoint) -> AnyPublisher<T, NetworkError> {  
        guard let url = endpoint.url else {  
            return Fail(error: .invalidURL).eraseToAnyPublisher()  
        }  
          
        var request = URLRequest(url: url)  
        request.httpMethod = endpoint.method  
        request.allHTTPHeaderFields = endpoint.headers  
          
        return session.dataTaskPublisher(for: request)  
            .tryMap { data, response -> Data in  
                guard let httpResponse = response as? HTTPURLResponse else {  
                    throw NetworkError.invalidResponse  
                }  
                guard (200...299).contains(httpResponse.statusCode) else {  
                    throw NetworkError.serverError(httpResponse.statusCode)  
                }  
                return data  
            }  
            .decode(type: T.self, decoder: decoder)  
            .mapError { error -> NetworkError in  
                if let networkError = error as? NetworkError {  
                    return networkError  
                }  
                return .decodingFailed(error)  
            }  
            .receive(on: DispatchQueue.main)  
            .eraseToAnyPublisher()  
    }  
      
    func requestData(_ endpoint: APIEndpoint) -> AnyPublisher<Data, NetworkError> {  
        guard let url = endpoint.url else {  
            return Fail(error: .invalidURL).eraseToAnyPublisher()  
        }  
          
        var request = URLRequest(url: url)  
        request.httpMethod = endpoint.method  
          
        return session.dataTaskPublisher(for: request)  
            .tryMap { data, response -> Data in  
                guard let httpResponse = response as? HTTPURLResponse,  
                      (200...299).contains(httpResponse.statusCode) else {  
                    throw NetworkError.invalidResponse  
                }  
                return data  
            }  
            .mapError { _ in NetworkError.invalidResponse }  
            .receive(on: DispatchQueue.main)  
            .eraseToAnyPublisher()  
    }  
}  
```  
  
### 3.2 APIEndpoint 协议  
  
```swift  
protocol APIEndpoint {  
    var baseURL: String { get }  
    var path: String { get }  
    var method: String { get }  
    var headers: [String: String]? { get }  
    var queryItems: [URLQueryItem]? { get }  
}  
  
extension APIEndpoint {  
    var url: URL? {  
        var components = URLComponents(string: baseURL + path)  
        components?.queryItems = queryItems  
        return components?.url  
    }  
      
    var headers: [String: String]? { nil }  
    var queryItems: [URLQueryItem]? { nil }  
}  
```

## 五、RSS 解析层：XMLParser  
  
原生 `XMLParser` 实现 RSS/Atom 解析，不依赖第三方库。  
  
```swift  
import Foundation  
  
struct RSSParsedItem {  
    let title: String  
    let link: String  
    let pubDate: Date?  
    let author: String?  
    let description: String?  
}  
  
final class RSSXMLParser: NSObject, XMLParserDelegate {  
    private var items: [RSSParsedItem] = []  
    private var currentElement = ""  
    private var currentTitle = ""  
    private var currentLink = ""  
    private var currentPubDate = ""  
    private var currentAuthor = ""  
    private var currentDescription = ""  
    private var insideItem = false  
      
    private let dateFormatter: DateFormatter = {  
        let formatter = DateFormatter()  
        formatter.locale = Locale(identifier: "en_US_POSIX")  
        formatter.dateFormat = "EEE, dd MMM yyyy HH:mm:ss Z"  
        return formatter  
    }()  
      
    func parse(data: Data) -> [RSSParsedItem] {  
        items.removeAll()  
        let parser = XMLParser(data: data)  
        parser.delegate = self  
        parser.parse()  
        return items  
    }  
      
    // MARK: - XMLParserDelegate  
      
    func parser(_ parser: XMLParser, didStartElement elementName: String, namespaceURI: String?, qualifiedName qName: String?, attributes attributeDict: [String : String] = [:]) {  
        currentElement = elementName  
          
        if elementName == "item" || elementName == "entry" {  
            insideItem = true  
            currentTitle = ""  
            currentLink = ""  
            currentPubDate = ""  
            currentAuthor = ""  
            currentDescription = ""  
        }  
          
        if elementName == "link", let href = attributeDict["href"] {  
            currentLink = href        // Atom format  
        }  
    }  
      
    func parser(_ parser: XMLParser, foundCharacters string: String) {  
        guard insideItem else { return }  
          
        let trimmed = string.trimmingCharacters(in: .whitespacesAndNewlines)  
        switch currentElement {  
        case "title": currentTitle += trimmed  
        case "link": currentLink += trimmed       // RSS 2.0  
        case "pubDate", "published", "updated": currentPubDate += trimmed  
        case "author", "name": currentAuthor += trimmed  
        case "description", "summary", "content": currentDescription += trimmed  
        default: break  
        }  
    }  
      
    func parser(_ parser: XMLParser, didEndElement elementName: String, namespaceURI: String?, qualifiedName qName: String?) {  
        if elementName == "item" || elementName == "entry" {  
            insideItem = false  
              
            let pubDate = dateFormatter.date(from: currentPubDate)  
              
            let item = RSSParsedItem(  
                title: currentTitle,  
                link: currentLink,  
                pubDate: pubDate,  
                author: currentAuthor.isEmpty ? nil : currentAuthor,  
                description: currentDescription.isEmpty ? nil : currentDescription  
            )  
            items.append(item)  
        }  
    }  
      
    func parser(_ parser: XMLParser, parseErrorOccurred parseError: Error) {  
        print("RSS parse error: \(parseError)")  
    }  
}  
```

## 六、安全层：Keychain 封装  
  
使用 `Security` 框架封装 API Key、Webhook URL 等敏感信息。  
  
```swift  
import Foundation  
import Security  
  
enum KeychainError: Error {  
    case itemNotFound  
    case duplicateItem  
    case invalidStatus(OSStatus)  
    case conversionFailed  
}  
  
final class KeychainManager {  
    static let shared = KeychainManager()  
    private init() {}  
      
    @discardableResult  
    func save(_ value: String, service: String, account: String) -> Bool {  
        guard let data = value.data(using: .utf8) else { return false }  
          
        delete(service: service, account: account)  
          
        let query: [String: Any] = [  
            kSecClass as String: kSecClassGenericPassword,  
            kSecAttrService as String: service,  
            kSecAttrAccount as String: account,  
            kSecValueData as String: data,  
            kSecAttrAccessible as String: kSecAttrAccessibleWhenUnlockedThisDeviceOnly  
        ]  
          
        let status = SecItemAdd(query as CFDictionary, nil)  
        return status == errSecSuccess  
    }  
      
    func retrieve(service: String, account: String) -> String? {  
        let query: [String: Any] = [  
            kSecClass as String: kSecClassGenericPassword,  
            kSecAttrService as String: service,  
            kSecAttrAccount as String: account,  
            kSecReturnData as String: true,  
            kSecMatchLimit as String: kSecMatchLimitOne  
        ]  
          
        var result: AnyObject?  
        let status = SecItemCopyMatching(query as CFDictionary, &result)  
          
        guard status == errSecSuccess,  
              let data = result as? Data,  
              let value = String(data: data, encoding: .utf8) else {  
            return nil  
        }  
        return value  
    }  
      
    @discardableResult  
    func delete(service: String, account: String) -> Bool {  
        let query: [String: Any] = [  
            kSecClass as String: kSecClassGenericPassword,  
            kSecAttrService as String: service,  
            kSecAttrAccount as String: account  
        ]  
        let status = SecItemDelete(query as CFDictionary)  
        return status == errSecSuccess || status == errSecItemNotFound  
    }  
}  
  
// MARK: - Convenience  
extension KeychainManager {  
    enum Key: String {  
        case aiAPIKey = "com.trendradar.ai_api_key"  
        case aiProvider = "com.trendradar.ai_provider"  
        case feishuWebhook = "com.trendradar.feishu_webhook"  
    }  
      
    func saveAIAPIKey(_ key: String, provider: String) {  
        save(key, service: Key.aiAPIKey.rawValue, account: provider)  
    }  
      
    func getAIAPIKey(for provider: String) -> String? {  
        retrieve(service: Key.aiAPIKey.rawValue, account: provider)  
    }  
}
```  

## 七、后台刷新：BGAppRefreshTask  
  
实现保活机制，后台静默抓取新热点，发现重大更新时推送本地通知。  
  
### 7.1 BackgroundTaskManager  
  
```swift  
import BackgroundTasks  
import UserNotifications  
  
final class BackgroundTaskManager {  
    static let shared = BackgroundTaskManager()  
    static let refreshTaskIdentifier = "com.trendradar.backgroundrefresh"  
      
    private let networkManager = NetworkManager.shared  
    private let dataController = DataController.shared  
      
    private init() {}  
      
    func registerTasks() {  
        BGTaskScheduler.shared.register(  
            forTaskWithIdentifier: Self.refreshTaskIdentifier,  
            using: nil  
        ) { [weak self] task in  
            self?.handleRefresh(task: task as! BGAppRefreshTask)  
        }  
    }  
      
    func scheduleRefresh() {  
        let request = BGAppRefreshTaskRequest(identifier: Self.refreshTaskIdentifier)  
        request.earliestBeginDate = Date(timeIntervalSinceNow: 15 * 60)  
          
        do {  
            try BGTaskScheduler.shared.submit(request)  
            print("Background refresh scheduled")  
        } catch {  
            print("Failed to schedule: \(error)")  
        }  
    }  
      
    private func handleRefresh(task: BGAppRefreshTask) {  
        scheduleRefresh()  
          
        let queue = OperationQueue()  
        queue.maxConcurrentOperationCount = 1  
          
        let refreshOperation = BlockOperation { [weak self] in  
            guard let self = self else { return }  
            self.performBackgroundFetch()  
        }  
          
        task.expirationHandler = {  
            queue.cancelAllOperations()  
        }  
          
        refreshOperation.completionBlock = {  
            task.setTaskCompleted(success: !refreshOperation.isCancelled)  
        }  
          
        queue.addOperations([refreshOperation], waitUntilFinished: false)  
    }  
      
    private func performBackgroundFetch() {  
        let platforms = ["zhihu", "weibo", "baidu", "toutiao"]  
        let group = DispatchGroup()  
        var newItemsCount = 0  
          
        for platform in platforms {  
            group.enter()  
            let endpoint = NewsNowEndpoint.hotlist(platform: platform)  
              
            networkManager.request(endpoint)  
                .sink(  
                    receiveCompletion: { _ in group.leave() },  
                    receiveValue: { [weak self] (response: NewsNowResponse) in  
                        let count = self?.processBackgroundResponse(response, platform: platform) ?? 0  
                        newItemsCount += count  
                    }  
                )  
                .store(in: &cancellables)  
        }  
          
        group.notify(queue: .main) { [weak self] in  
            if newItemsCount > 0 {
```  

### 8.1 RadarViewModel（发现页）  
  
```swift  
import Foundation  
import Combine  
import SwiftData  
  
@MainActor  
final class RadarViewModel: ObservableObject {  
    @Published var hotlist: [HotNewsItem] = []  
    @Published var filteredItems: [HotNewsItem] = []  
    @Published var selectedPlatform: String? = nil  
    @Published var isLoading: Bool = false  
    @Published var errorMessage: String?  
    @Published var lastUpdated: Date?  
      
    private let networkManager = NetworkManager.shared  
    private let dataController = DataController.shared  
    private var cancellables = Set<AnyCancellable>()  
      
    private var context: ModelContext { dataController.context }  
      
    func fetchHotlist(forceRefresh: Bool = false) {  
        guard !isLoading else { return }  
        isLoading = true  
        errorMessage = nil  
          
        let endpoint = NewsNowEndpoint.hotlist(platform: selectedPlatform)  
          
        networkManager.request(endpoint)  
            .flatMap { [weak self] (response: NewsNowResponse) -> AnyPublisher<[HotNewsItem], NetworkError> in  
                guard let self = self else {  
                    return Just([]).setFailureType(to: NetworkError.self).eraseToAnyPublisher()  
                }  
                return self.processAndSave(response: response)  
            }  
            .receive(on: DispatchQueue.main)  
            .sink(  
                receiveCompletion: { [weak self] completion in  
                    self?.isLoading = false  
                    if case .failure(let error) = completion {  
                        self?.errorMessage = error.localizedDescription  
                        self?.loadFromLocalCache()  
                    }  
                },  
                receiveValue: { [weak self] items in  
                    self?.hotlist = items  
                    self?.applyKeywordFilter()  
                    self?.lastUpdated = Date()  
                }  
            )  
            .store(in: &cancellables)  
    }  
      
    func applyKeywordFilter() {  
        let descriptor = FetchDescriptor<UserKeyword>(predicate: #Predicate { $0.isActive == true })  
        guard let keywords = try? context.fetch(descriptor) else {  
            filteredItems = hotlist  
            return  
        }  
          
        if keywords.isEmpty {  
            filteredItems = hotlist  
            return  
        }  
          
        filteredItems = hotlist.filter { item in  
            keywords.contains { keyword in  
                switch keyword.type {  
                case .include:  
                    return item.title.localizedCaseInsensitiveContains(keyword.word)  
                case .required:  
                    return item.title.localizedCaseInsensitiveContains(keyword.word)  
                case .exclude:  
                    return !item.title.localizedCaseInsensitiveContains(keyword.word)  
                }  
            }  
        }  
    }  
      
    func markAsRead(_ item: HotNewsItem) {  
        item.isRead = true  
        try? context.save()  
        objectWillChange.send()  
    }  
      
    func toggleBookmark(_ item: HotNewsItem) {  
        item.isBookmarked.toggle()  
        try? context.save()  
        objectWillChange.send()  
    }  
      
    private func processAndSave(response: NewsNowResponse) -> AnyPublisher<[HotNewsItem], NetworkError> {  
        Future<[HotNewsItem], NetworkError> { [weak self] promise in  
            guard let self = self else {  
                promise(.success([]))  
                return  
            }  
              
            let items = response.data.map { dto -> HotNewsItem in  
                let id = "\(dto.title.hashValue)_\(dto.source)"  
                let descriptor = FetchDescriptor<HotNewsItem>(predicate: #Predicate { $0.id == id })  
                let existing = try? self.context.fetch(descriptor).first  
                  
                if let existing = existing {  
                    existing.lastSeenAt = Date()  
                    existing.previousRank = existing.currentRank  
                    existing.currentRank = dto.rank  
                    existing.mentionCount += 1  
                    existing.isNew = false  
                      
                    let snapshot = TrendSnapshot(  
                        timestamp: Date(),  
                        rank: dto.rank,  
                        platformId: dto.source,  
                        newsItem: existing  
                    )  
                    self.context.insert(snapshot)  
                      
                    return existing  
                } else {  
                    let newItem = HotNewsItem(  
                        id: id,  
                        title: dto.title,  
                        url: dto.url,  
                        mobileUrl: dto.mobileUrl,  
                        platformId: dto.source,  
                        platformName: dto.sourceName,  
                        currentRank: dto.rank,  
                        isNew: true  
                    )  
                    self.context.insert(newItem)  
                    return newItem  
                }  
            }  
              
            try? self.context.save()  
            promise(.success(items))  
        }  
        .eraseToAnyPublisher()  
    }  
      
    private func loadFromLocalCache() {  
        let descriptor = FetchDescriptor<HotNewsItem>(  
            sortBy: [SortDescriptor(\.lastSeenAt, order: .reverse)]  
        )  
        hotlist = (try? context.fetch(descriptor)) ?? []  
        applyKeywordFilter()  
    }  
}
```  

```swift  
final class AIService {  
    static let shared = AIService()  
    private let networkManager = NetworkManager.shared  
      
    private var apiKey: String? {  
        KeychainManager.shared.getAIAPIKey(for: currentProvider)  
    }  
      
    private var currentProvider: String {  
        UserDefaults.standard.string(forKey: "ai_provider") ?? "deepseek"  
    }  
      
    private init() {}  
      
    func analyzeTrends(items: [HotNewsItem]) -> AnyPublisher<AIAnalysisRecord, NetworkError> {  
        guard let apiKey = apiKey else {  
            return Fail(error: NetworkError.invalidURL).eraseToAnyPublisher()  
        }  
          
        let prompt = buildAnalysisPrompt(items: items)  
        let endpoint = AIEndpoint.chat(provider: currentProvider, apiKey: apiKey, prompt: prompt)  
          
        return networkManager.request(endpoint)  
            .map { [weak self] (response: AIChatResponse) -> AIAnalysisRecord in  
                self?.parseAIResponse(response, items: items) ?? AIAnalysisRecord(  
                    reportMode: "current",  
                    trendOverview: "Analysis failed",  
                    sentimentPositive: 0,  
                    sentimentNeutral: 0,  
                    sentimentNegative: 0,  
                    strategyRecommendation: ""  
                )  
            }  
            .eraseToAnyPublisher()  
    }  
      
    private func buildAnalysisPrompt(items: [HotNewsItem]) -> String {  
        let titles = items.prefix(50).map { "\($0.currentRank). [\($0.platformName)] \($0.title)" }.joined(separator: "\n")  
          
        return """  
        You are a professional sentiment analyst. Analyze the following hot news and output JSON:  
          
        News list:  
        \(titles)  
          
        Output fields:  
        - trend_overview: overview of key hot topics (within 200 chars)  
        - sentiment_positive: positive sentiment ratio (0.0-1.0)  
        - sentiment_neutral: neutral sentiment ratio (0.0-1.0)  
        - sentiment_negative: negative sentiment ratio (0.0-1.0)  
        - weak_signals: array of weak signals and anomalies  
        - strategy_recommendation: strategic advice (within 200 chars)  
        """  
    }  
      
    private func parseAIResponse(_ response: AIChatResponse, items: [HotNewsItem]) -> AIAnalysisRecord {  
        let content = response.choices.first?.message.content ?? ""  
        let record = AIAnalysisRecord(  
            reportMode: "current",  
            trendOverview: content.prefix(200).description,  
            sentimentPositive: 0.4,  
            sentimentNeutral: 0.4,  
            sentimentNegative: 0.2,  
            strategyRecommendation: "Keep watching",  
            rawResponse: content  
        )  
        record.relatedItems = items  
        return record  
    }  
}
```  

## 十二、性能与体验优化  
  
| 优化点 | 实现方式 |  
|--------|---------|  
| **列表性能** | `LazyVStack` + `ForEach` with `id`，避免全量加载 |  
| **图片加载** | `AsyncImage` (iOS 15+) 原生异步加载，自带缓存 |  
| **数据去重** | 以 `hash(title + platform)` 为唯一键，更新时比对 |  
| **增量更新** | 后台抓取只拉取变更，UI 局部刷新 |  
| **离线优先** | 先展示 SwiftData 缓存，网络返回后 Diff 更新 |  
| **触觉反馈** | `UIImpactFeedbackGenerator` 在收藏/刷新时触发 |  
| **骨架屏** | `redacted(reason: .placeholder)` 实现加载占位 |

## 十三、三阶段实施计划

### 13.1 执行规则

本方案按三个阶段逐项实施。每个任务完成后需要满足代码实现、测试验证、GitHub Actions 构建和 IPA 产物检查四项条件，再进入下一项。

任务状态说明：

| 状态 | 含义 |
|---|---|
| 已完成 | 当前代码已经具备并经过构建验证 |
| 已落地 | 当前代码已实现，后续补充测试或体验优化 |
| 待实施 | 已确定实施，等待开发 |
| 需确认 | 需要产品取舍、外部数据或额外权限，等待用户决定 |
| 建议保留 | 对产品完整性有帮助，按当前方案保留 |
| 建议暂缓 | 当前数据或系统条件不足，放入后续阶段 |

每个任务使用以下验收标准：

1. 功能可以在 iPhone 端完成闭环操作。
2. 数据保存、错误处理和空状态完整。
3. 关键逻辑有单元测试或集成测试。
4. GitHub Actions 成功编译并生成标准 Payload 结构的无签名 IPA。
